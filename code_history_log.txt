// ─── Watch Folders List (Settings Tab) ───────────────────────────────────────
function renderWatchFoldersList() {
  const container = document.getElementById('watch-folders-list');
  if (!container) return;
  container.innerHTML = '';

  const folders = currentSettings.watchFolders || [];

  if (folders.length === 0) {
    container.innerHTML = `<div class="empty-folders-hint" style="color:var(--text-muted); padding:10px; font-size:12px;"><i class="fa-solid fa-circle-info"></i> No folders configured. Add one below.</div>`;
    return;
  }

  folders.forEach((folderPath, index) => {
    const item = document.createElement('div');
    item.className = 'watch-folder-item';
    item.innerHTML = `
      <div class="folder-item-icon">

      </div>

function updateDashboardFolders() {
  const el = document.getElementById('stat-watch-path');
  if (!el) return;
  const folders = currentSettings.watchFolders || [];
  if (folders.length === 0) {
    el.textContent = 'No folders configured';
  } else if (folders.length === 1) {
    el.textContent = folders[0];
  } else {
    el.textContent = `${folders.length} folders monitored`;
    el.title = folders.join('\n');
  }
}

// ─── Category Cards & Rules ──────────────────────────────────────────────────

  const metadata = currentSettings.categoryMetadata || {};

  Object.keys(categories).forEach(category => {

    const meta = metadata[category] || {};
    const iconClass = meta.icon || defaultIcons[category] || 'fa-folder';

    const isMisc = category === 'Misc';

    let cardHtml = `
      <div class="category-card-header">

    if (!isMisc) {

            `).join('')}

            <div class="add-tag-group">
              <input type="text" class="input-add-tag" id="input-add-${category}" placeholder="e.g. .ts" onkeydown="handleTagInputKeydown(event, '${category}')">
              <button class="btn-add-tag" onclick="addExtension('${category}')">

function getPreviewPathTemplate(category) {
  const tmpl = currentSettings.folderTemplate || 'standard';
  switch (tmpl) {
    case 'clean': return `[Root] / ${category} / file.ext`;
    case 'category-ext': return `[Root] / ${category} / [ext] / file.ext`;
    case 'date-category': return `[Root] / [Date] / ${category} / file.ext`;
    case 'year-month-category': return `[Root] / [YYYY] / [Month] / ${category} / file.ext`;
    case 'custom': return `[Root] / ${currentSettings.customTemplateString || '{category}/{ext}'} / file.ext`;
    case 'standard':
    default: return `[Root] / [Date] / ${category} / [ext] / file.ext`;
  }
}

window.removeExtension = (category, ext) => {
  if (!currentSettings) return;
  const index = currentSettings.categories[category].indexOf(ext);
  if (index > -1) {
    currentSettings.categories[category].splice(index, 1);
    renderCategoryCards();
    updateStats();
    saveSettingsSilently();
  }
};

window.addExtension = (category) => {
  if (!currentSettings) return;
  const input = document.getElementById(`input-add-${category}`);
  let ext = input.value.trim().toLowerCase();
  if (!ext) return;
  if (!ext.startsWith('.')) ext = '.' + ext;

  let exists = false;
  Object.keys(currentSettings.categories).forEach(cat => {
    if (currentSettings.categories[cat].includes(ext)) exists = true;
  });

  if (exists) {
    alert(`Extension "${ext}" is already configured!`);
    input.value = '';
    return;
  }

  currentSettings.categories[category].push(ext);
  input.value = '';
  renderCategoryCards();
  updateStats();
  saveSettingsSilently();
};

window.handleTagInputKeydown = (event, category) => {
  if (event.key === 'Enter') {
    event.preventDefault();
    window.addExtension(category);
  }
};

window.deleteCategory = (category) => {
  if (!currentSettings || !confirm(`Delete custom category "${category}"?`)) return;
  delete currentSettings.categories[category];
  delete currentSettings.categoryMetadata[category];
  renderCategoryCards();
  updateStats();
  saveSettingsSilently();
};

// ─── Modal for New Category ──────────────────────────────────────────────────
let selectedModalIcon = 'fa-folder';
let selectedModalColor = 'primary-gradient';

function setupModal() {
  const modal = document.getElementById('modal-create-category');
  const openBtn = document.getElementById('btn-open-create-category');
  const closeBtn = document.getElementById('btn-close-cat-modal');
  const cancelBtn = document.getElementById('btn-cancel-cat-modal');
  const saveBtn = document.getElementById('btn-save-new-category');

  if (!modal) return;

  const openModal = () => {
    modal.classList.add('open');
    document.getElementById('modal-input-cat-name').value = '';
    document.getElementById('modal-input-cat-exts').value = '';
  };
  const closeModal = () => modal.classList.remove('open');

  openBtn.addEventListener('click', openModal);
  closeBtn.addEventListener('click', closeModal);
  cancelBtn.addEventListener('click', closeModal);

  // Icon picker
  document.querySelectorAll('#modal-icon-picker .icon-option').forEach(opt => {
    opt.addEventListener('click', () => {
      document.querySelectorAll('#modal-icon-picker .icon-option').forEach(o => o.classList.remove('selected'));
      opt.classList.add('selected');
      selectedModalIcon = opt.getAttribute('data-icon');
    });
  });

  // Color picker
  document.querySelectorAll('#modal-color-picker .color-option').forEach(opt => {
    opt.addEventListener('click', () => {
      document.querySelectorAll('#modal-color-picker .color-option').forEach(o => o.classList.remove('selected'));
      opt.classList.add('selected');
      selectedModalColor = opt.getAttribute('data-color');
    });
  });

  // Save new category
  saveBtn.addEventListener('click', () => {
    const nameInput = document.getElementById('modal-input-cat-name');
    const extsInput = document.getElementById('modal-input-cat-exts');
    const catName = nameInput.value.trim();

    if (!catName) {
      alert('Please enter a category name.');
      return;
    }

    if (currentSettings.categories[catName]) {
      alert(`Category "${catName}" already exists.`);
      return;
    }

    const exts = extsInput.value
      .split(',')
      .map(e => e.trim().toLowerCase())
      .filter(Boolean)
      .map(e => e.startsWith('.') ? e : '.' + e);

    currentSettings.categories[catName] = exts;
    currentSettings.categoryMetadata[catName] = {
      icon: selectedModalIcon,
      color: selectedModalColor,
      custom: true
    };

    closeModal();
    renderCategoryCards();
    updateStats();
    saveSettingsSilently();
  });
}

// ─── Category Distribution Breakdown Bar ────────────────────────────────────
function renderDistributionBar() {
  const barTrack = document.getElementById('dist-bar-track');
  const legendGrid = document.getElementById('dist-legend-grid');
  const totalLabel = document.getElementById('dist-total-label');
  if (!barTrack || !legendGrid) return;

  const categories = currentSettings ? currentSettings.categories : {};
  const metadata = currentSettings ? currentSettings.categoryMetadata : {};

  // Count success logs by category
  const counts = {};
  Object.keys(categories).forEach(c => counts[c] = 0);
  let totalOrganized = 0;

  currentLogs.forEach(l => {
    if (l.status === 'success') {
      totalOrganized++;
      const cat = l.category || 'Misc';
      counts[cat] = (counts[cat] || 0) + 1;
    }
  });

  if (totalLabel) totalLabel.textContent = `${totalOrganized} files organized`;

  barTrack.innerHTML = '';
  legendGrid.innerHTML = '';

  const catNames = Object.keys(counts);

  if (totalOrganized === 0) {
    barTrack.innerHTML = `<div class="dist-bar-segment slate-gradient" style="width:100%; opacity:0.3;"></div>`;
    catNames.slice(0, 6).forEach(cat => {
      const meta = metadata[cat] || {};
      const color = meta.color || defaultColors[cat] || 'slate-gradient';
      const item = document.createElement('div');
      item.className = 'dist-legend-item';
      item.innerHTML = `<span class="dist-legend-dot ${color}"></span> ${cat}: <span class="dist-legend-count">0 (0%)</span>`;
      legendGrid.appendChild(item);
    });
    return;
  }

  catNames.forEach(cat => {
    const count = counts[cat] || 0;
    if (count > 0 || catNames.length <= 6) {
      const pct = Math.round((count / totalOrganized) * 100);
      const meta = metadata[cat] || {};
      const color = meta.color || defaultColors[cat] || 'slate-gradient';

      if (count > 0) {
        const seg = document.createElement('div');
        seg.className = `dist-bar-segment ${color}`;
        seg.style.width = `${(count / totalOrganized) * 100}%`;
        seg.title = `${cat}: ${count} files (${pct}%)`;
        barTrack.appendChild(seg);
      }

      const leg = document.createElement('div');
      leg.className = 'dist-legend-item';
      leg.innerHTML = `<span class="dist-legend-dot ${color}"></span> ${cat}: <span class="dist-legend-count">${count} (${pct}%)</span>`;
      legendGrid.appendChild(leg);
    }
  });
}

function renderLogs() {
  const tableBody = document.getElementById('logs-table-body');
  const recentLogsContainer = document.getElementById('dashboard-recent-logs');

  const filteredLogs = currentLogs.filter(log => {
    if (activeFilter !== 'all' && log.status !== activeFilter) return false;
    if (searchKeyword) return (log.message || '').toLowerCase().includes(searchKeyword);
    return true;
  });

  tableBody.innerHTML = '';
  if (filteredLogs.length === 0) {
    tableBody.innerHTML = `
      <tr class="empty-row">
        <td colspan="4">
          <div class="empty-state">
            <i class="fa-solid fa-clipboard-list"></i>
            <p>No matching logs found.</p>
          </div>
        </td>
      </tr>
    `;
  } else {
    filteredLogs.forEach(log => {
      const row = document.createElement('tr');
      const canOpen = log.destination || log.source;
      row.innerHTML = `
        <td class="timestamp-col">${formatLogTime(log.timestamp)}</td>
        <td>
          <span class="log-status-badge ${log.status}">
            ${log.status === 'success' ? '<i class="fa-solid fa-circle-check"></i>' : ''}
            ${log.status === 'info'    ? '<i class="fa-solid fa-circle-info"></i>'  : ''}
            ${log.status === 'error'   ? '<i class="fa-solid fa-circle-exclamation"></i>' : ''}
            ${log.status}
          </span>
        </td>
        <td>${log.message}</td>
        <td style="text-align:right;">
          <div class="log-row-actions" style="justify-content:flex-end;">
            ${canOpen ? `
              <button class="btn-table-action" onclick="openPathInFolder('${(log.destination || log.source).replace(/\\/g, '\\\\')}')" title="Open containing folder">
                <i class="fa-regular fa-folder-open"></i>
              </button>
            ` : ''}
            ${log.status === 'success' && log.destination ? `
              <button class="btn-table-action" onclick="undoSpecificMove('${(log.destination).replace(/\\/g, '\\\\')}')" title="Undo this file move" style="color:var(--accent-cyan);">
                <i class="fa-solid fa-rotate-left"></i>
              </button>
            ` : ''}
          </div>
        </td>
      `;
      tableBody.appendChild(row);
    });
  }

  // Dashboard live panel – last 5
  recentLogsContainer.innerHTML = '';
  const recentLogs = currentLogs.filter(l => l.status === 'success' || l.status === 'error').slice(0, 5);

  if (recentLogs.length === 0) {
    recentLogsContainer.innerHTML = `
      <div class="empty-state">
        <i class="fa-solid fa-magnifying-glass"></i>
        <p>No recent file operations yet.</p>
      </div>
    `;
  } else {
    recentLogs.forEach(log => {
      const logDiv = document.createElement('div');
      logDiv.className = 'live-log-item';
      logDiv.innerHTML = `
        <span class="live-log-dot ${log.status}"></span>
        <div class="live-log-content">
          <p class="live-log-msg">${log.message}</p>
          <span class="live-log-time">${formatLogTime(log.timestamp, true)}</span>
        </div>
      `;
      recentLogsContainer.appendChild(logDiv);
    });
  }
}

window.openPathInFolder = (targetPath) => {
  ipcRenderer.invoke('open-path', targetPath);
};

window.undoSpecificMove = async (dstPath) => {
  const history = await ipcRenderer.invoke('get-undo-history');
  const match = history.find(h => h.destinationPath === dstPath && !h.undone);
  if (match) {
    const res = await ipcRenderer.invoke('undo-move', match.id);
    if (res.success) {
      alert(`Restored "${res.fileName}" back to original location.`);
    } else {
      alert(`Undo failed: ${res.message}`);
    }
  } else {
    alert('No active undo record found for this file.');
  }
};

function formatLogTime(isoString, timeOnly = false) {
  try {
    const date = new Date(isoString);
    const timeParts = date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
    if (timeOnly) return timeParts;
    const dateParts = date.toLocaleDateString([], { month: 'short', day: '2-digit' });
    return `${dateParts} ${timeParts}`;
  } catch {
    return isoString;
  }
}

function updateStats() {
  const successLogs = currentLogs.filter(log => log.status === 'success');
  document.getElementById('stat-files-count').textContent = successLogs.length;
  document.getElementById('stat-folders-count').textContent =
    (currentSettings && currentSettings.watchFolders) ? currentSettings.watchFolders.length : 0;

  if (currentSettings && currentSettings.categories) {
    const catKeys = Object.keys(currentSettings.categories);
    document.getElementById('stat-categories-count').textContent = catKeys.length;
    let totalExts = 0;
    catKeys.forEach(k => totalExts += (currentSettings.categories[k] || []).length);
    document.getElementById('stat-exts-count').textContent = `${totalExts} total extensions`;
  }

  const lastLog = currentLogs.find(l => l.status === 'success');
  if (lastLog) {
    document.getElementById('stat-last-action').textContent = (lastLog.category || 'Organized') + ' File';
    document.getElementById('stat-last-time').textContent = formatLogTime(lastLog.timestamp);
  } else {
    document.getElementById('stat-last-action').textContent = 'Idle';
    document.getElementById('stat-last-time').textContent = 'Awaiting files';
  }
}

    const btn = document.getElementById(btnId);

    btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> Organizing...';

    await ipcRenderer.invoke('open-tracker-file');

  document.getElementById('sim-input-filename').addEventListener('input', runSimulation);

  document.getElementById('input-log-search').addEventListener('input', (e) => {
    searchKeyword = e.target.value.toLowerCase().trim();
    renderLogs();
  });

  // Log filter badges
  const filterBadges = document.querySelectorAll('.filter-badge');
  filterBadges.forEach(badge => {
    badge.addEventListener('click', () => {
      filterBadges.forEach(b => b.classList.remove('active'));
      badge.classList.add('active');
      activeFilter = badge.getAttribute('data-filter');
      renderLogs();
    });
  });
}

// ─── Drive & System Safety Guard ─────────────────────────────────────────────
function isDriveOrSystemRoot(folderPath) {
  if (!folderPath) return true;
  const resolved = path.resolve(folderPath).toLowerCase().replace(/\//g, '\\');
  // Match drive roots like "c:\", "a:\", "d:\"
  if (/^[a-z]:\\?$/.test(resolved)) return true;
  // Match critical system directories
  const sysDirs = ['c:\\windows', 'c:\\program files', 'c:\\program files (x86)', 'c:\\users'];
  if (sysDirs.includes(resolved)) return true;
  return false;
}

// ─── Watcher Management (multi-folder) ──────────────────────────────────────
function stopAllWatchers() {
  for (const w of watchers) {
    try { w.close(); } catch (e) { /* ignore */ }
  }
  watchers = [];
}

function startWatcher() {
  stopAllWatchers();

  const settings = settingsManager.getSettings();
  const watchFolders = settings.watchFolders || [];

  if (watchFolders.length === 0) {
    settingsManager.logActivity('info', 'No folders configured to watch.');
    return;
  }

  const stability = (settings.stabilityDelay || 2) * 1000;

  for (const watchPath of watchFolders) {
    // Block drive roots & system paths
    if (isDriveOrSystemRoot(watchPath)) {
      settingsManager.logActivity('error', `Blocked: Cannot watch system or drive root "${watchPath}".`);
      continue;
    }

    if (!fs.existsSync(watchPath)) {
      settingsManager.logActivity('error', `Folder not found: "${watchPath}"`);
      continue;
    }

    settingsManager.logActivity('info', `Monitoring: "${watchPath}"`);

    const w = chokidar.watch(watchPath, {
      depth: 0,
      ignoreInitial: true,
      usePolling: true,
      interval: 200,
      awaitWriteFinish: {
        stabilityThreshold: stability,
        pollInterval: 100
      }
    });

    w.on('add', (filePath) => {
      if (watcherActive) {
        // Small delay to ensure any file handle is unlocked
        setTimeout(() => {
          processFile(filePath, watchPath);
        }, 300);
      }
    });

    w.on('error', (error) => {
      settingsManager.logActivity('error', `Watcher error on "${watchPath}": ${error.message}`);
    });

    watchers.push(w);
  }
}

function stopWatcher() {
  stopAllWatchers();
  settingsManager.logActivity('info', 'All file watchers stopped.');
}

function toggleWatcher() {
  watcherActive = !watcherActive;
  updateTrayMenu();

  if (watcherActive) {
    startWatcher();
    settingsManager.logActivity('info', 'File organizer resumed.');
  } else {
    stopWatcher();
    settingsManager.logActivity('info', 'File organizer paused.');
  }

  if (mainWindow) {
    mainWindow.webContents.send('watcher-status-changed', watcherActive);
  }
}

// ─── Category / Extension Detection ─────────────────────────────────────────
function getCategoryForFile(filename, settings) {
  const ext = path.extname(filename).toLowerCase();
  const cats = settings.categories || {};

  for (const [category, extensions] of Object.entries(cats)) {
    if (category === 'Misc') continue;
    if (Array.isArray(extensions) && extensions.includes(ext)) {
      return category;
    }
  }

  return 'Misc';
}

// ─── Date Formatting Helper ──────────────────────────────────────────────────
function formatFolderDate(dateObj, formatType) {
  const date = dateObj instanceof Date && !isNaN(dateObj) ? dateObj : new Date();
  const day = String(date.getDate()).padStart(2, '0');
  const dayNum = date.getDate();
  const monthNames = [
    'January', 'February', 'March', 'April', 'May', 'June',
    'July', 'August', 'September', 'October', 'November', 'December'
  ];
  const shortMonthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const monthName = monthNames[date.getMonth()];
  const shortMonthName = shortMonthNames[date.getMonth()];
  const year = date.getFullYear();

  let formattedDate = `${dayNum} ${monthName} ${year}`; // default

  switch (formatType) {
    case 'YYYY-MM-DD':
      formattedDate = `${year}-${month}-${day}`;
      break;
    case 'DD-MM-YYYY':
      formattedDate = `${day}-${month}-${year}`;
      break;
    case 'YYYY/MM':
      formattedDate = path.join(String(year), month);
      break;
    case 'Month YYYY':
      formattedDate = `${monthName} ${year}`;
      break;
    case 'DD MMMM YYYY':
    default:
      formattedDate = `${dayNum} ${monthName} ${year}`;
      break;
  }

  return {
    formattedDate,
    year,
    month,
    monthName,
    shortMonthName,
    day: dayNum
  };
}

// ─── Wildcard Matching Helper ────────────────────────────────────────────────
function matchWildcard(str, pattern) {
  if (!pattern) return false;
  const regexPattern = '^' + pattern.replace(/[-[\]/{}()+?.\\^$|]/g, '\\$&').replace(/\*/g, '.*') + '$';
  try {
    return new RegExp(regexPattern, 'i').test(str);
  } catch (e) {
    return false;
  }
}

// ─── Destination Path Builder ────────────────────────────────────────────────
function computeDestination(filePath, watchRoot, settings) {
  const baseName = path.basename(filePath);
  const ext = path.extname(filePath).toLowerCase();
  const extName = ext.replace('.', '') || 'misc';
  const category = getCategoryForFile(baseName, settings);

  // Target base root
  let targetRoot = watchRoot;
  if (settings.targetMode === 'custom-folder' && settings.customTargetFolder && fs.existsSync(settings.customTargetFolder)) {
    targetRoot = settings.customTargetFolder;
  }

  // Date selection
  let targetDate = new Date();
  if (settings.dateSource === 'fileDate') {
    try {
      const stat = fs.statSync(filePath);
      targetDate = stat.birthtime || stat.mtime || targetDate;
    } catch (e) {}
  }

  const { formattedDate, year, month, monthName, day } = formatFolderDate(targetDate, settings.dateFormat);

  let subDir = '';
  const template = settings.folderTemplate || 'standard';

  switch (template) {
    case 'clean':
      subDir = category;
      break;
    case 'category-ext':
      subDir = path.join(category, extName);
      break;
    case 'date-category':
      subDir = path.join(formattedDate, category);
      break;
    case 'year-month-category':
      subDir = path.join(String(year), `${month} - ${monthName}`, category);
      break;
    case 'custom':
      let custom = settings.customTemplateString || '{category}/{ext}';
      custom = custom
        .replace(/{category}/gi, category)
        .replace(/{ext}/gi, extName)
        .replace(/{date}/gi, formattedDate)
        .replace(/{year}/gi, String(year))
        .replace(/{month}/gi, String(month))
        .replace(/{monthname}/gi, monthName)
        .replace(/{day}/gi, String(day));
      subDir = custom.split(/[\/\\]+/).filter(Boolean).join(path.sep);
      break;
    case 'standard':
    default:
      subDir = path.join(formattedDate, category, extName);
      break;
  }

  const targetDir = path.join(targetRoot, subDir);

  // Filename formatting
  let finalFileName = baseName;
  const rawExt = path.extname(finalFileName);
  const rawBase = path.basename(finalFileName, rawExt);

  let processedBase = rawBase;
  let processedExt = settings.lowercaseExtensions ? rawExt.toLowerCase() : rawExt;

  if (settings.cleanFileNames) {
    processedBase = processedBase.replace(/[\s\-_]+/g, '_').replace(/[^a-zA-Z0-9_\(\)\.]/g, '');
  }

  if (settings.autoDatePrefix) {
    const prefix = `${year}-${month}-${String(day).padStart(2, '0')}_`;
    if (!processedBase.startsWith(prefix)) {
      processedBase = prefix + processedBase;
    }
  }

  finalFileName = processedBase + processedExt;

  return {
    targetDir,
    targetFileName: finalFileName,
    initialTargetFilePath: path.join(targetDir, finalFileName),
    category,
    extName,
    formattedDate
  };
}

// ─── Duplicate Resolution ────────────────────────────────────────────────────
function resolveDuplicatePath(targetFilePath, duplicateHandling, sourceFilePath) {
  if (!fs.existsSync(targetFilePath)) {
    return { targetPath: targetFilePath, action: 'create' };
  }

  if (duplicateHandling === 'skip') {
    return { targetPath: null, action: 'skip' };
  }

  if (duplicateHandling === 'overwrite') {
    return { targetPath: targetFilePath, action: 'overwrite' };
  }

  if (duplicateHandling === 'smart-match') {
    try {
      const srcStat = fs.statSync(sourceFilePath);
      const dstStat = fs.statSync(targetFilePath);
      if (srcStat.size === dstStat.size) {
        return { targetPath: null, action: 'skip_identical' };
      }
    } catch (e) {}
  }

  // Default: rename
  const dirName = path.dirname(targetFilePath);
  const ext = path.extname(targetFilePath);
  const baseName = path.basename(targetFilePath, ext);

  let counter = 1;
  let newPath = targetFilePath;

  while (fs.existsSync(newPath)) {
    newPath = path.join(dirName, `${baseName}(${counter})${ext}`);
    counter++;
  }

  return { targetPath: newPath, action: 'rename' };
}

function logToTrackerFile(fileName, category, extName, finalTargetFilePath) {
  try {

  }
}

// ─── Safe File Move with Retry ───────────────────────────────────────────────
async function safeMoveFile(src, dst, maxRetries = 3) {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      fs.ensureDirSync(path.dirname(dst));
      fs.moveSync(src, dst, { overwrite: true });
      return true;
    } catch (err) {
      if ((err.code === 'EBUSY' || err.code === 'EPERM') && attempt < maxRetries) {
        await new Promise(r => setTimeout(r, 600 * attempt));
      } else {
        throw err;
      }
    }
  }
  return false;
}

// ─── Core File Processor ─────────────────────────────────────────────────────
async function processFile(filePath, watchRoot) {
  try {
    if (!fs.existsSync(filePath)) return;

    // Block files directly under drive roots
    const parentDir = path.dirname(path.resolve(filePath));
    if (isDriveOrSystemRoot(parentDir)) return;

    const stats = fs.statSync(filePath);
    if (!stats.isFile()) return;

    const baseName = path.basename(filePath);
    const ext = path.extname(filePath).toLowerCase();

    // Skip hidden files, system files, desktop.ini
    if (baseName.startsWith('.') || baseName === 'desktop.ini' || baseName === 'thumbs.db') return;

    const settings = settingsManager.getSettings();

    // Excluded extensions
    const excludedExts = settings.excludedExtensions || [];
    if (excludedExts.map(e => e.toLowerCase()).includes(ext)) return;

    // Excluded filename patterns
    const excludedPatterns = settings.excludedFilenamePatterns || [];
    for (const pat of excludedPatterns) {
      if (matchWildcard(baseName, pat)) return;
    }

    // Size filters
    if (settings.minFileSizeKB > 0 && (stats.size / 1024) < settings.minFileSizeKB) return;
    if (settings.maxFileSizeMB > 0 && (stats.size / (1024 * 1024)) > settings.maxFileSizeMB) return;

    const computed = computeDestination(filePath, watchRoot, settings);

    // Don't move a file into itself
    const resolvedSrc = path.resolve(filePath);
    const resolvedDst = path.resolve(computed.initialTargetFilePath);
    if (resolvedSrc === resolvedDst) return;

    const dupResult = resolveDuplicatePath(computed.initialTargetFilePath, settings.duplicateHandling, filePath);

    if (!dupResult.targetPath) {
      settingsManager.logActivity('info', `Skipped duplicate/identical: "${baseName}"`);
      return;
    }

    const finalTargetFilePath = dupResult.targetPath;
    const finalBaseName = path.basename(finalTargetFilePath);

    // Perform the safe move with retry
    await safeMoveFile(filePath, finalTargetFilePath);

    // Record for Undo history
    settingsManager.recordMoveForUndo({
      sourcePath: filePath,
      destinationPath: finalTargetFilePath,
      fileName: baseName,
      category: computed.category,
      ext: computed.extName
    });

    // Write to tracker text file
    logToTrackerFile(baseName, computed.category, computed.extName, finalTargetFilePath);

    // Write to internal activity log
    const relDst = path.relative(watchRoot, finalTargetFilePath);
    settingsManager.logActivity('success', `Moved: "${baseName}" → "${relDst}"`, {
      source: filePath,
      destination: finalTargetFilePath,
      category: computed.category,
      ext: computed.extName
    });

    if (settings.notificationsEnabled !== false) {
      showNotification('File Organized', `"${baseName}" → ${computed.category}/${computed.extName}`);
    }

    if (mainWindow) {
      mainWindow.webContents.send('activity-log-update', settingsManager.getLogs());
    }
  } catch (error) {
    console.error('File organization error:', error);
    settingsManager.logActivity('error', `Failed to organize "${path.basename(filePath)}": ${error.message}`);
  }
}

// ─── Manual Scan (all watch folders) ────────────────────────────────────────
async function organizeExistingFiles() {
  try {
    const settings = settingsManager.getSettings();
    const watchFolders = settings.watchFolders || [];

    if (watchFolders.length === 0) {
      settingsManager.logActivity('info', 'No folders configured. Add folders in Settings.');
      return;
    }

    let totalCount = 0;

    for (const watchPath of watchFolders) {
      if (isDriveOrSystemRoot(watchPath) || !fs.existsSync(watchPath)) continue;

      settingsManager.logActivity('info', `Scanning: "${watchPath}"...`);

      const entries = fs.readdirSync(watchPath);

      for (const fileName of entries) {
        const filePath = path.join(watchPath, fileName);
        try {
          const stat = fs.statSync(filePath);
          if (!stat.isFile()) continue;

          if (fileName.startsWith('.') || fileName === 'desktop.ini' || fileName === 'thumbs.db') continue;

          await processFile(filePath, watchPath);
          totalCount++;
        } catch (e) { /* skip locked/inaccessible files */ }
      }
    }

    settingsManager.logActivity('info', `Manual scan complete. Processed ${totalCount} file(s).`);
    showNotification('Scan Complete', `Processed ${totalCount} file(s) across ${watchFolders.length} folder(s).`);

    if (mainWindow) {
      mainWindow.webContents.send('activity-log-update', settingsManager.getLogs());
    }
  } catch (error) {
    settingsManager.logActivity('error', `Manual scan failed: ${error.message}`);
  }
}

// ─── Undo Move Operation ─────────────────────────────────────────────────────
async function undoMoveOperation(undoId) {
  try {
    const history = settingsManager.getUndoHistory();
    const item = undoId ? history.find(h => h.id === undoId) : history.find(h => !h.undone);

    if (!item) {
      return { success: false, message: 'No reversible operation found.' };
    }

    if (item.undone) {
      return { success: false, message: 'This operation has already been undone.' };
    }

    if (!fs.existsSync(item.destinationPath)) {
      return { success: false, message: `File at "${item.destinationPath}" is no longer present.` };
    }

    // Ensure original directory exists
    fs.ensureDirSync(path.dirname(item.sourcePath));

    // Move back
    await safeMoveFile(item.destinationPath, item.sourcePath);

    settingsManager.markUndoComplete(item.id);
    settingsManager.logActivity('info', `Undone: Restored "${item.fileName}" back to "${item.sourcePath}"`);

    if (mainWindow) {
      mainWindow.webContents.send('activity-log-update', settingsManager.getLogs());
    }

    return { success: true, fileName: item.fileName, sourcePath: item.sourcePath };
  } catch (err) {
    return { success: false, message: err.message };
  }
}

// ─── Drive Root Guard ────────────────────────────────────────────────────────
function isDriveRoot(folderPath) {
  const resolved = path.resolve(folderPath).toLowerCase().replace(/\//g, '\\');
  // Match patterns like "c:\", "a:\", "d:\", etc.
  return /^[a-z]:\\$/.test(resolved) || /^[a-z]:$/.test(resolved);
}

// ─── Watcher Management (multi-folder) ──────────────────────────────────────
function stopAllWatchers() {
  for (const w of watchers) {
    try { w.close(); } catch (e) { /* ignore */ }
  }
  watchers = [];
}

function startWatcher() {
  stopAllWatchers();

  const settings = settingsManager.getSettings();
  const watchFolders = settings.watchFolders || [];

  if (watchFolders.length === 0) {
    settingsManager.logActivity('info', 'No folders configured to watch.');
    return;
  }

  for (const watchPath of watchFolders) {
    // Block drive roots
    if (isDriveRoot(watchPath)) {
      settingsManager.logActivity('error', `Blocked: Cannot watch drive root "${watchPath}".`);
      continue;
    }

    if (!fs.existsSync(watchPath)) {
      settingsManager.logActivity('error', `Folder not found: "${watchPath}"`);
      continue;
    }

    settingsManager.logActivity('info', `Monitoring: "${watchPath}"`);

    const w = chokidar.watch(watchPath, {
      depth: 0,
      ignoreInitial: true,
      usePolling: true,
      interval: 150,
      awaitWriteFinish: {
        stabilityThreshold: 2000,
        pollInterval: 100
      }
    });

    w.on('add', (filePath) => {
      if (watcherActive) {
        processFile(filePath, watchPath);
      }
    });

    w.on('error', (error) => {
      settingsManager.logActivity('error', `Watcher error on "${watchPath}": ${error.message}`);
    });

    watchers.push(w);
  }
}

function stopWatcher() {
  stopAllWatchers();
  settingsManager.logActivity('info', 'All file watchers stopped.');
}

function toggleWatcher() {
  watcherActive = !watcherActive;
  updateTrayMenu();

  if (watcherActive) {
    startWatcher();
    settingsManager.logActivity('info', 'File organizer resumed.');
  } else {
    stopWatcher();
    settingsManager.logActivity('info', 'File organizer paused.');
  }

  if (mainWindow) {
    mainWindow.webContents.send('watcher-status-changed', watcherActive);
  }
}

// ─── Category / Extension Detection ─────────────────────────────────────────
function getCategoryForFile(filename) {
  const ext = path.extname(filename).toLowerCase();
  const settings = settingsManager.getSettings();

  for (const [category, extensions] of Object.entries(settings.categories)) {
    if (extensions.includes(ext)) {
      return category;
    }
  }

  return 'Misc';
}

// ─── Date Formatting ─────────────────────────────────────────────────────────
function getFormattedDate() {
  const date = new Date();
  const day = date.getDate();
  const monthNames = [
    'January', 'February', 'March', 'April', 'May', 'June',
    'July', 'August', 'September', 'October', 'November', 'December'
  ];
  const month = monthNames[date.getMonth()];
  const year = date.getFullYear();
  return `${day} ${month} ${year}`;
}

// ─── Duplicate Resolution ────────────────────────────────────────────────────
function resolveDuplicatePath(targetFilePath, duplicateHandling) {
  if (!fs.existsSync(targetFilePath)) {
    return targetFilePath;
  }

  if (duplicateHandling === 'skip') {
    return null;
  }

  const dirName = path.dirname(targetFilePath);
  const ext = path.extname(targetFilePath);
  const baseName = path.basename(targetFilePath, ext);

  let counter = 1;
  let newPath = targetFilePath;

  while (fs.existsSync(newPath)) {
    newPath = path.join(dirName, `${baseName}(${counter})${ext}`);
    counter++;
  }

  return newPath;
}

// ─── Text Log File ───────────────────────────────────────────────────────────
function logToTrackerFile(fileName, category, extName, finalTargetFilePath) {
  try {
    const documentsPath = app.getPath('documents');
    const txtFilePath = path.join(documentsPath, 'background organizer.txt');

    const now = new Date();
    const timestamp = now.toLocaleString('en-IN', {
      day: '2-digit', month: 'short', year: 'numeric',
      hour: '2-digit', minute: '2-digit', second: '2-digit'
    });

    const line = `[${timestamp}]  Name: ${fileName}  |  Type: ${category} / ${extName}  |  Location: ${finalTargetFilePath}\n`;
    fs.appendFileSync(txtFilePath, line, 'utf8');
  } catch (error) {
    console.error('Failed to write to background organizer.txt:', error.message);
  }
}

// ─── Core File Processor ─────────────────────────────────────────────────────
function processFile(filePath, watchRoot) {
  try {
    if (!fs.existsSync(filePath)) return;

    // Block files directly under drive roots
    const parentDir = path.dirname(path.resolve(filePath));
    if (isDriveRoot(parentDir)) return;

    const stats = fs.statSync(filePath);
    if (!stats.isFile()) return;

    const baseName = path.basename(filePath);

    // Skip hidden files, system files, and temp downloads
    if (baseName.startsWith('.') || baseName === 'desktop.ini' || baseName === 'thumbs.db') return;

    const ext = path.extname(filePath).toLowerCase();
    const tempExtensions = ['.crdownload', '.tmp', '.part', '.download', '.partial', '.!bt'];
    if (tempExtensions.includes(ext)) return;

    const extName = ext.replace('.', '') || 'unknown';
    const settings = settingsManager.getSettings();
    const category = getCategoryForFile(baseName);
    const dateFolder = getFormattedDate();

    // Build target path: [WatchRoot] / [Date] / [Category] / [Extension] / [File]
    let targetDir = watchRoot;
    if (settings.dateWiseFolder) {
      targetDir = path.join(targetDir, dateFolder);
    }
    targetDir = path.join(targetDir, category, extName);

    // Don't move a file into its own parent (already organized)
    const resolvedSrc = path.resolve(filePath);
    const resolvedDst = path.resolve(path.join(targetDir, baseName));
    if (resolvedSrc === resolvedDst) return;

    fs.ensureDirSync(targetDir);

    const initialTargetFilePath = path.join(targetDir, baseName);
    const finalTargetFilePath = resolveDuplicatePath(initialTargetFilePath, settings.duplicateHandling);

    if (!finalTargetFilePath) {
      settingsManager.logActivity('info', `Skipped duplicate: ${baseName}`);
      return;
    }

    const finalBaseName = path.basename(finalTargetFilePath);

    // Move the file
    fs.moveSync(filePath, finalTargetFilePath);

    // Write to tracker text file
    logToTrackerFile(baseName, category, extName, finalTargetFilePath);

    // Write to internal activity log
    const relPath = `${settings.dateWiseFolder ? dateFolder + '/' : ''}${category}/${extName}/${finalBaseName}`;
    settingsManager.logActivity('success', `Moved: "${baseName}" → "${relPath}"`);
    showNotification('File Organized', `"${baseName}" → ${category}/${extName}`);

    if (mainWindow) {
      mainWindow.webContents.send('activity-log-update', settingsManager.getLogs());
    }
  } catch (error) {
    console.error('File organization error:', error);
    settingsManager.logActivity('error', `Failed to organize "${path.basename(filePath)}": ${error.message}`);
  }
}

// ─── Manual Scan (all watch folders) ────────────────────────────────────────
function organizeExistingFiles() {
  try {
    const settings = settingsManager.getSettings();
    const watchFolders = settings.watchFolders || [];

    if (watchFolders.length === 0) {
      settingsManager.logActivity('info', 'No folders configured. Add folders in Settings.');
      return;
    }

    let totalCount = 0;

    for (const watchPath of watchFolders) {
      if (isDriveRoot(watchPath) || !fs.existsSync(watchPath)) continue;

      settingsManager.logActivity('info', `Scanning: "${watchPath}"...`);

      const entries = fs.readdirSync(watchPath);

      for (const fileName of entries) {
        const filePath = path.join(watchPath, fileName);
        try {
          const stat = fs.statSync(filePath);
          if (!stat.isFile()) continue;

          if (fileName.startsWith('.') || fileName === 'desktop.ini') continue;

          const ext = path.extname(filePath).toLowerCase();
          const tempExtensions = ['.crdownload', '.tmp', '.part', '.download', '.partial', '.!bt'];
          if (tempExtensions.includes(ext)) continue;

          processFile(filePath, watchPath);
          totalCount++;
        } catch (e) { /* skip locked/inaccessible files */ }
      }
    }

    settingsManager.logActivity('info', `Manual scan complete. Processed ${totalCount} file(s).`);
    showNotification('Scan Complete', `Processed ${totalCount} file(s) across ${watchFolders.length} folder(s).`);

    if (mainWindow) {
      mainWindow.webContents.send('activity-log-update', settingsManager.getLogs());
    }
  } catch (error) {
    settingsManager.logActivity('error', `Manual scan failed: ${error.message}`);
  }
}

// ─── Load Settings ────────────────────────────────────────────────────────────
function loadSettings() {
  try {
    if (fs.existsSync(settingsFilePath)) {
      const data = fs.readJsonSync(settingsFilePath);
      const defaults = getDefaults();

      currentSettings = { ...defaults, ...data };

      // Migrate old single watchFolder to watchFolders array
      if (data.watchFolder && !data.watchFolders) {
        currentSettings.watchFolders = [data.watchFolder];
        delete currentSettings.watchFolder;
      }

      // Ensure watchFolders is always an array
      if (!Array.isArray(currentSettings.watchFolders)) {
        currentSettings.watchFolders = [defaults.watchFolders[0]];
      }

      // Deep merge categories (keep defaults + user additions, no duplicates)
      currentSettings.categories = {};
      const userCategories = data.categories || {};
      for (const [cat, defaultExts] of Object.entries(defaults.categories)) {
        const userExts = userCategories[cat] || [];
        currentSettings.categories[cat] = [...new Set([...defaultExts, ...userExts])];
      }

      // Include any user-added custom categories
      for (const [cat, exts] of Object.entries(userCategories)) {
        if (!currentSettings.categories[cat]) {
          currentSettings.categories[cat] = exts;
        }
      }
    } else {
      currentSettings = getDefaults();
      saveSettings(currentSettings);
    }
  } catch (error) {
    console.error('Failed to load settings:', error);
    currentSettings = getDefaults();
  }

  return currentSettings;
}

// ─── Save Settings ────────────────────────────────────────────────────────────
function saveSettings(settings) {
  try {
    currentSettings = settings;
    fs.writeJsonSync(settingsFilePath, currentSettings, { spaces: 2 });
    return true;
  } catch (error) {
    console.error('Failed to save settings:', error);
    logActivity('error', `Failed to save settings: ${error.message}`);
    return false;
  }
}

function getSettings() {
  if (!currentSettings) return getDefaults();
  return currentSettings;
}

// ─── Activity Log ────────────────────────────────────────────────────────────
function logActivity(status, message) {
  try {
    const timestamp = new Date().toISOString();
    const logLine = JSON.stringify({ timestamp, status, message }) + '\n';

    fs.appendFileSync(logFilePath, logLine, 'utf8');

    // Trim log to last 1000 lines if it grows too large
    const logContent = fs.readFileSync(logFilePath, 'utf8');
    const lines = logContent.trim().split('\n');
    if (lines.length > 2000) {
      const keepLines = lines.slice(lines.length - 1000);
      fs.writeFileSync(logFilePath, keepLines.join('\n') + '\n', 'utf8');
    }

    return { timestamp, status, message };
  } catch (error) {
    console.error('Failed to write activity log:', error);
    return null;
  }
}

function getLogs() {
  try {
    if (!fs.existsSync(logFilePath)) return [];
    const logContent = fs.readFileSync(logFilePath, 'utf8').trim();
    if (!logContent) return [];

    return logContent
      .split('\n')
      .filter(Boolean)
      .map(line => {
        try {
          return JSON.parse(line);
        } catch {
          return { timestamp: new Date().toISOString(), status: 'info', message: line };
        }
      })
      .reverse();
  } catch (error) {
    console.error('Failed to read logs:', error);
    return [];
  }
}

function clearLogs() {
  try {
    fs.writeFileSync(logFilePath, '', 'utf8');
    logActivity('info', 'Activity log cleared.');
    return true;
  } catch (error) {
    console.error('Failed to clear logs:', error);
    return false;
  }
}

// ─── Tab Navigation ──────────────────────────────────────────────────────────
function setupTabs() {
  const navItems = document.querySelectorAll('.nav-item');
  const tabContents = document.querySelectorAll('.tab-content');

  navItems.forEach(item => {
    item.addEventListener('click', (e) => {
      e.preventDefault();
      const tabId = item.getAttribute('data-tab');
      navItems.forEach(n => n.classList.remove('active'));
      item.classList.add('active');
      tabContents.forEach(tab => {
        tab.id === tabId ? tab.classList.add('active') : tab.classList.remove('active');
      });
    });
  });
}

// ─── Load Settings ────────────────────────────────────────────────────────────
async function loadSettings() {
  currentSettings = await ipcRenderer.invoke('get-settings');

  // Ensure watchFolders is always an array
  if (!Array.isArray(currentSettings.watchFolders)) {
    currentSettings.watchFolders = currentSettings.watchFolder ? [currentSettings.watchFolder] : [];
  }

  renderWatchFoldersList();
  updateDashboardFolders();

  const modeEl = document.getElementById('mode-move');
  if (modeEl) modeEl.checked = true;

  document.getElementById('toggle-date-wise').checked = currentSettings.dateWiseFolder;

  if (currentSettings.duplicateHandling === 'rename') {
    document.getElementById('dup-rename').checked = true;
  } else {
    document.getElementById('dup-skip').checked = true;
  }

  document.getElementById('toggle-startup').checked = currentSettings.startupLaunch;
  document.getElementById('toggle-organize-on-launch').checked = currentSettings.organizeOnLaunch;

  renderCategoryCards();
  updateStats();
}

// ─── Watch Folders List (Settings Tab) ───────────────────────────────────────
function renderWatchFoldersList() {
  const container = document.getElementById('watch-folders-list');
  if (!container) return;
  container.innerHTML = '';

  const folders = currentSettings.watchFolders || [];

  if (folders.length === 0) {
    container.innerHTML = `<div class="empty-folders-hint"><i class="fa-solid fa-circle-info"></i> No folders configured. Add one below.</div>`;
    return;
  }

  folders.forEach((folderPath, index) => {
    const item = document.createElement('div');
    item.className = 'watch-folder-item';
    item.innerHTML = `
      <div class="folder-item-icon">
        <i class="fa-solid fa-folder-open"></i>
      </div>
      <div class="folder-item-path" title="${folderPath}">${folderPath}</div>
      <button class="folder-item-remove" onclick="removeWatchFolder(${index})" title="Remove this folder">
        <i class="fa-solid fa-xmark"></i>
      </button>
    `;
    container.appendChild(item);
  });
}

// Update dashboard monitored path display
function updateDashboardFolders() {
  const el = document.getElementById('stat-watch-path');
  if (!el) return;
  const folders = currentSettings.watchFolders || [];
  if (folders.length === 0) {
    el.textContent = 'No folders configured';
  } else if (folders.length === 1) {
    el.textContent = folders[0];
  } else {
    el.textContent = `${folders.length} folders monitored`;
    el.title = folders.join('\n');
  }
}

  if (!currentSettings) return;
  currentSettings.watchFolders.splice(index, 1);
  renderWatchFoldersList();
  updateDashboardFolders();
  saveSettingsSilently();
};

// ─── Extension Tag Helpers ────────────────────────────────────────────────────
window.removeExtension = (category, ext) => {
  if (!currentSettings) return;
  const index = currentSettings.categories[category].indexOf(ext);
  if (index > -1) {
    currentSettings.categories[category].splice(index, 1);
    renderCategoryCards();
    saveSettingsSilently();
  }
};

window.addExtension = (category) => {
  if (!currentSettings) return;
  const input = document.getElementById(`input-add-${category}`);
  if (!input) return;
  let ext = input.value.trim();
  if (!ext) return;

  // Auto-prepend '.' if missing
  if (!ext.startsWith('.')) ext = '.' + ext;

  if (!currentSettings.categories[category].includes(ext)) {
    currentSettings.categories[category].push(ext);
    renderCategoryCards();
    saveSettingsSilently();
  }
  input.value = '';
};

window.handleTagInputKeydown = (event, category) => {
  if (event.key === 'Enter') {
    event.preventDefault();
    window.addExtension(category);
  }
};

// ─── Logs ────────────────────────────────────────────────────────────────────
async function loadLogs() {
  currentLogs = await ipcRenderer.invoke('get-logs');
  renderLogs();
  updateStats();
}

function renderLogs() {
  const tbody = document.getElementById('logs-table-body');
  if (!tbody) return;
  tbody.innerHTML = '';

  let filteredLogs = currentLogs;

  // Status filter
  if (activeFilter !== 'all') {
    filteredLogs = filteredLogs.filter(log => log.status === activeFilter);
  }

  // Search filter
  if (searchKeyword) {
    filteredLogs = filteredLogs.filter(log =>
      (log.message && log.message.toLowerCase().includes(searchKeyword)) ||
      (log.status && log.status.toLowerCase().includes(searchKeyword)) ||
      (log.timestamp && log.timestamp.toLowerCase().includes(searchKeyword))
    );
  }

  if (filteredLogs.length === 0) {
    tbody.innerHTML = `
      <tr class="empty-row">
        <td colspan="3">
          <div class="empty-state">
            <i class="fa-solid fa-clipboard-list"></i>
            <p>No matching log entries.</p>
          </div>
        </td>
      </tr>
    `;
    return;
  }

  filteredLogs.forEach(log => {
    const tr = document.createElement('tr');
    tr.className = `log-row log-${log.status}`;

    const timestamp = log.timestamp ? new Date(log.timestamp).toLocaleString() : '—';
    const statusIcon = log.status === 'success' ? 'fa-check-circle' :
                       log.status === 'error' ? 'fa-circle-exclamation' :
                       'fa-circle-info';

    tr.innerHTML = `
      <td class="log-timestamp">${timestamp}</td>
      <td><span class="log-status-badge status-${log.status}"><i class="fa-solid ${statusIcon}"></i> ${log.status}</span></td>
      <td class="log-message">${log.message || '—'}</td>
    `;

    tbody.appendChild(tr);
  });
}

// ─── Stats ───────────────────────────────────────────────────────────────────
function updateStats() {
  // Update files count
  const successCount = (currentLogs || []).filter(log => log.status === 'success').length;
  const filesEl = document.getElementById('stat-files-count');
  if (filesEl) filesEl.textContent = successCount;

  // Update folder count
  const folders = (currentSettings && currentSettings.watchFolders) || [];
  const foldersEl = document.getElementById('stat-folders-count');
  if (foldersEl) foldersEl.textContent = folders.length;
}

// ─── Preset Folder Helpers ────────────────────────────────────────────────────
function getPresetFolderPath(presetName) {
  // Uses path.sep and resolves to the user's home directory
  const home = require('os').homedir();
  const sep = require('path').sep;
  const presetMap = {
    'Downloads': 'Downloads',
    'Desktop': 'Desktop',
    'Documents': 'Documents',
    'Pictures': 'Pictures'
  };
  const relative = presetMap[presetName];
  return relative ? home + sep + relative : null;
}

// ─── Event Listeners ─────────────────────────────────────────────────────────
function setupEventListeners() {
  // Browse Folder
  document.getElementById('btn-browse-folder').addEventListener('click', async () => {
    const folder = await ipcRenderer.invoke('select-folder', currentSettings.watchFolders[0]);
    if (folder) {
      if (!currentSettings.watchFolders.includes(folder)) {
        currentSettings.watchFolders.push(folder);
        renderWatchFoldersList();
        updateDashboardFolders();
        saveSettingsSilently();
      } else {
        alert('This folder is already being monitored.');
      }
    }
  });

  // Preset folder buttons
  document.querySelectorAll('.preset-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      const presetName = btn.getAttribute('data-path');
      const folderPath = getPresetFolderPath(presetName);
      if (folderPath && !currentSettings.watchFolders.includes(folderPath)) {
        currentSettings.watchFolders.push(folderPath);
        renderWatchFoldersList();
        updateDashboardFolders();
        saveSettingsSilently();
      } else if (folderPath) {
        alert('This folder is already being monitored.');
      }
    });
  });

  // Duplicate handling radio buttons
  document.querySelectorAll('input[name="duplicate-handling"]').forEach(radio => {
    radio.addEventListener('change', (e) => {
      if (e.target.checked) {
        currentSettings.duplicateHandling = e.target.value;
        saveSettingsSilently();
      }
    });
  });

  // Date-wise toggle
  document.getElementById('toggle-date-wise').addEventListener('change', (e) => {
    currentSettings.dateWiseFolder = e.target.checked;
    saveSettingsSilently();
  });

  // Startup toggle
  document.getElementById('toggle-startup').addEventListener('change', (e) => {
    currentSettings.startupLaunch = e.target.checked;
    saveSettingsSilently();
  });

  // Organize on launch toggle
  document.getElementById('toggle-organize-on-launch').addEventListener('change', (e) => {
    currentSettings.organizeOnLaunch = e.target.checked;
    saveSettingsSilently();
  });

  // Clear logs
  document.getElementById('btn-clear-logs').addEventListener('click', async () => {
    await ipcRenderer.invoke('clear-logs');
    currentLogs = [];
    renderLogs();
    updateStats();
  });

  // Toggle watcher service from sidebar
  document.getElementById('btn-toggle-service').addEventListener('click', async () => {
    const isActive = await ipcRenderer.invoke('toggle-watcher');
    updateWatcherStatusUI(isActive);
  });

  // Manual Scan
  const triggerManualScan = async (btnId) => {
    const btn = document.getElementById(btnId);
    const originalHtml = btn.innerHTML;
    btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> Organizing...';
    btn.disabled = true;
    await ipcRenderer.invoke('trigger-manual-scan');
    setTimeout(() => {
      btn.disabled = false;
      btn.innerHTML = '<i class="fa-solid fa-check"></i> Complete!';
      setTimeout(() => { btn.innerHTML = originalHtml; }, 2000);
    }, 1000);
  };

  document.getElementById('btn-manual-scan-dash').addEventListener('click', () => triggerManualScan('btn-manual-scan-dash'));
  document.getElementById('btn-manual-scan-settings').addEventListener('click', () => triggerManualScan('btn-manual-scan-settings'));

  // Open watch folder (opens first in list, or explorer)
  document.getElementById('btn-open-watch-folder').addEventListener('click', () => {
    const folders = currentSettings && currentSettings.watchFolders;
    if (folders && folders.length > 0) {
      shell.openPath(folders[0]).catch(err => alert('Could not open folder: ' + err.message));
    } else {
      alert('No folder configured.');
    }
  });

  // Open tracker text file
  document.getElementById('btn-open-tracker').addEventListener('click', async () => {
    await ipcRenderer.invoke('open-tracker-file');
  });

  // Logs search
  document.getElementById('input-log-search').addEventListener('input', (e) => {
    searchKeyword = e.target.value.toLowerCase().trim();
    renderLogs();
  });

  // Log filter badges
  const filterBadges = document.querySelectorAll('.filter-badge');
  filterBadges.forEach(badge => {
    badge.addEventListener('click', () => {
      filterBadges.forEach(b => b.classList.remove('active'));
      badge.classList.add('active');
      activeFilter = badge.getAttribute('data-filter');
      renderLogs();
    });
  });
}
